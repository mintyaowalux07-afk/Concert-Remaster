ตั้งค่าเชื่อมต่อ Google Sheets

บันทึกทุกคำตอบเข้า Google Sheets ทันทีอัตโนมัติ (Real-time)

เชื่อมต่อ Google Sheets พร้อมทำงานแล้ว

ขั้นตอนการเชื่อมต่อใน 2 นาที (ทำครั้งเดียว):

1
สร้าง Google Sheets เปล่า

เปิด sheets.new เพื่อสร้าง Sheet ใหม่ จากนั้นไปที่เมนูด้านบน ส่วนขยาย (Extensions) > Apps Script

2
คัดลอกโค้ด Apps Script ไปวาง

ลบโค้ดเดิมในหน้า Apps Script แล้วนำโค้ดด้านล่างนี้ไปวางแทนทั้งหมด

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      "รหัส Voucher", "ชื่อ-นามสกุล", "เบอร์โทร", "อีเมล", 
      "บริษัท", "ตำแหน่ง", "ช่องทางที่รู้จัก", "คะแนนรวม (1-5)", 
      "เนื้อหา", "สถานที่", "เจ้าหน้าที่", "NPS (0-10)", 
      "ความสนใจ", "การติดต่อกลับ", "ข้อเสนอแนะ", "วันเวลาบันทึก"
    ]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    data.id, data.name, data.phone, data.email, 
    data.company, data.role, data.source, data.overallRating, 
    data.contentScore, data.venueScore, data.staffScore, data.nps, 
    (data.interests || []).join(", "), data.contactConsent, data.feedback, data.createdAt
  ]);
  return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}
3
Deploy เป็น Web App

กดปุ่มสีฟ้า ทำให้ใช้งานได้ (Deploy) > การทำให้ใช้งานได้รายการใหม่ (New deployment) > เลือกประเภท เว็บแอป (Web app) > ช่อง Who has access ให้เลือกเป็น ทุกคน (Anyone) แล้วกด Deploy จากนั้น Copy Web app URL มาใส่ในช่องด้านบน
